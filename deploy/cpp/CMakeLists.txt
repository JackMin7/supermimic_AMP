cmake_minimum_required(VERSION 3.15)
project(wmrl_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(pybind11 CONFIG REQUIRED)
find_package(Threads REQUIRED)
pybind11_add_module(wmrl_transport transport.cpp)
target_include_directories(wmrl_transport PRIVATE /usr/local/include /usr/local/include/ddscxx)
target_link_libraries(wmrl_transport PRIVATE /usr/local/lib/libunitree_sdk2.a /usr/local/lib/libddscxx.so /usr/local/lib/libddsc.so Threads::Threads dl)
set_target_properties(wmrl_transport PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "/usr/local/lib" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
